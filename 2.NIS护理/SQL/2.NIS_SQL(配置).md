## 5.导管记录新增单科室

```sql
INSERT INTO nis_plugin_conf (
    id,
    department,
    parent_id,
    type,
    sort,
    name,
    icon,
    class_full_name,
    method,
    form_name,
    web_url,
    is_visible,
    mini_is_visible,
    record_child_type,
    countsql,
    webpopup
)
SELECT 
    TO_CHAR(NVL((SELECT MAX(TO_NUMBER(id)) FROM nis_plugin_conf ), 0) + 1),
    '43200', -- 换成对应的科室
    parent_id,
    type,
    sort,
    name,
    icon,
    class_full_name,
    method,
    form_name,
    web_url,
    is_visible,
    mini_is_visible,
    record_child_type,
    countsql,
    webpopup
FROM nis_plugin_conf
WHERE department = '43100' AND name = '导管记录';
```
