## 1. PDA移动护理医嘱列表

```sql
SELECT 
	e.* 
FROM 
    view_met_nui_exerecord e
WHERE 1=1 
	AND e.valid_flag = '1'  -- 只查询有效记录  
    AND e.inpatient_no LIKE '%501161'  -- 住院流水号
ORDER BY 
    CASE WHEN e.exe_state = '0' THEN 0 WHEN e.exe_state = '1' THEN 1 ELSE 2 END,  -- 待执行排最前
    e.use_time ASC, 
    e.exec_sqn;
```

## 6.触发器批量执行跟新医嘱查询里面

```sql
CREATE OR REPLACE TRIGGER trg_update_nurse_confirm
AFTER INSERT ON nis.met_nui_exerecord
FOR EACH ROW
when (NEW.is_scan = '5')
DECLARE
    v_ipm_order_id      NUMBER;
    v_inpatient_no      NVARCHAR2(16);
    v_record_count      NUMBER;
BEGIN
    -- 1. 更新 MET_IPM_EXECUNDRUG 表
    UPDATE met_ipm_execundrug g
    SET --g.exec_flag = '1',
        g.exec_date = :NEW.END_TIME,
        g.exec_usercd = :NEW.END_OPER
    WHERE g.exec_sqn = :NEW.exec_sqn;

    -- 2. 查找关联的 ipm_order_id 和 inpatient_no
    SELECT TO_NUMBER(r.mark6), r.inpatient_no
    INTO v_ipm_order_id, v_inpatient_no
    FROM met_ipm_execundrug g
    LEFT JOIN met_ipm_order r ON g.mo_order = r.mo_order
    WHERE g.exec_sqn = :NEW.exec_sqn
      AND ROWNUM = 1;

    -- 3. 检查 ord_ipm_nurse_confirm 表中是否已有记录
    SELECT COUNT(*)
    INTO v_record_count
    FROM ord_ipm_nurse_confirm
    WHERE ipm_order_id = v_ipm_order_id;

    IF v_record_count > 0 THEN
        -- 已有记录，执行 UPDATE
        UPDATE ord_ipm_nurse_confirm m
        SET m.bgn_time = TO_CHAR(:NEW.END_TIME, 'YYYY-MM-DD HH24:MI:SS'),
            m.opr_time = TO_CHAR(:NEW.END_TIME, 'YYYY-MM-DD HH24:MI:SS'),
            m.execflag = '1',
            m.BGN_NURSE_ID = :NEW.END_OPER
        WHERE m.ipm_order_id = v_ipm_order_id;
    ELSE
        -- 没有记录，执行 INSERT
        INSERT INTO ord_ipm_nurse_confirm (
            ipm_order_id,           -- 住院医嘱ID
            bgn_time,               -- 医嘱执行时间
            dc_nurse_id,            -- 停止医嘱执行护士（可为空）
            dc_time,                -- 停止医嘱执行时间（可为空）
            opr_id,                 -- 操作员ID
            opr_time,               -- 操作时间
            bgn_nurse_id,           -- 医嘱执行护士
            inpatient_no,           -- HIS患者住院流水号
            bgn_double_nurse_id,    -- 双签名（可为空）
            check_time,             -- 双签名时间（可为空）
            confirm_nurse_id,       -- 核准人（可为空）
            confirm_date,           -- 核准时间（可为空）
            confirmstop_nurse_id,   -- 停止核准人（可为空）
            confirmstop_date,       -- 停止核准时间（可为空）
            execreason,             -- 签名原因（可为空）
            execflag                -- 执行状态 1-已执行
        ) VALUES (
            v_ipm_order_id,                                         -- ipm_order_id
            TO_CHAR(:NEW.END_TIME, 'YYYY-MM-DD HH24:MI:SS'),       -- bgn_time
            NULL,                                                   -- dc_nurse_id
            NULL,                                                   -- dc_time
            :NEW.END_OPER,                                          -- opr_id（操作员ID设为执行人）
            TO_CHAR(:NEW.END_TIME, 'YYYY-MM-DD HH24:MI:SS'),       -- opr_time
            :NEW.END_OPER,                                          -- bgn_nurse_id
            v_inpatient_no,                                         -- inpatient_no
            NULL,                                                   -- bgn_double_nurse_id
            NULL,                                                   -- check_time
            NULL,                                                   -- confirm_nurse_id
            NULL,                                                   -- confirm_date
            NULL,                                                   -- confirmstop_nurse_id
            NULL,                                                   -- confirmstop_date
            NULL,                                                   -- execreason
            '1'                                                     -- execflag
        );
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        NULL;
    WHEN VALUE_ERROR THEN
        NULL;
    WHEN OTHERS THEN
        RAISE_APPLICATION_ERROR(-20001, '触发器执行错误: ' || SQLERRM);
END trg_update_nurse_confirm;

```
