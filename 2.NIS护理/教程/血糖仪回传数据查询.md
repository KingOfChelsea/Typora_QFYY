# 血糖仪回传数据查询

## 1.NIS自查数据

`NIS文书`自查`sql`

```sql
SELECT a.*, a.ROWID FROM nsr_nis_record_detail a
where a.inpatient_no = 'ZY0010000524442' 
and a.code = 'B80' 
and a.oper_type = 18; -- 血糖仪上传标识

select d.ID,
       d.RECORD_ID,
       d.INPATIENT_NO,
       d.RECORD_TIME,
       d.CODE,
       d.VALUE,
       d.SEQ_NO,
       d.DATA_SOURCE,
       d.OPER_CODE,
       d.OPER_TIME,
       d.OPER_NAME,
       d.OPER_TYPE,
       d.R_NAME,
       d.SHOW_TEMPERATURE,
       d.SHOW_NURSE_RECORD,
       d.GUID,
       d.oper_type,(case
                     when d.record_id is not null then
                      (SELECT nnr.SUMMARY_TITLE
                         FROM NSR_NIS_RECORD nnr
                        WHERE nnr.ID = d.RECORD_ID)
                     else
                      null
                   end) as SUMMARY_TITLE,
       (case
         when d.record_id is not null then
          (SELECT nnr.show_classify
             FROM NSR_NIS_RECORD nnr
            WHERE nnr.ID = d.RECORD_ID)
         else
          null
       end) as showClassify,
       d.OPER_CONTENT
  from nsr_nis_record_detail d
 where RECORD_ID in (2997480,
                     2999277,
                     3000692,
                     3002306,
                     3007166,
                     3007167,
                     3007168,
                     3009105,
                     3009129,
                     3009519,
                     3016927,
                     3016928,
                     3016935,
                     3018510,
                     3020391,
                     3027023,
                     3028648,
                     3030341)
```
